# WOL-Server - settings for the automatic setup on the first boot
#
# This file is on the boot partition of the SD card ("bootfs"). On the first boot the Pi
# applies the values and then deletes the file, because it contains passwords.
# The result is then in wol-setup.log on the same partition.
#
# Format: KEY=value (no spaces around the "="). Empty value = default.

# Name on the network. The GUI will be reachable at http://<HOSTNAME>.local/.
HOSTNAME=wol

# Time zone, important for the schedules
TIMEZONE=Europe/Berlin

# Wi-Fi country (allowed channels and transmit power)
COUNTRY=DE

# Password for the web GUI (at least 8 characters). Strongly recommended: without a password,
# anyone within radio range can use the GUI via the open hotspot on the first boot.
GUI_PASSWORD=

# Optional: preset the Wi-Fi. Leave empty = set up via the hotspot "WOL-Setup-XXXX".
WIFI_SSID=
WIFI_PASSWORD=

# Low-power mode: HDMI and Bluetooth off.
LOWPOWER=yes

# Maintenance access. Default: no login, the device does not need one.
# SSH=yes enables the SSH server. Log in with a key from the file
# ssh_authorized_keys (place it next to this file) or with ADMIN_PASSWORD.
# ADMIN_USER=root: maintenance directly as root - via SSH then only with a key.
ADMIN_USER=admin
ADMIN_PASSWORD=
SSH=no

# OpenVPN: place the client configuration as "openvpn.ovpn" next to this file,
# credentials (if needed) as "openvpn.auth" (line 1: user, line 2: password).
# The Pi installs OpenVPN as soon as it has Internet via Wi-Fi.
